#include <Keypad.h>

// Mendefinisikan pin untuk LED
const int redPin = 13;   // Pin untuk LED merah
const int greenPin = 12; // Pin untuk LED hijau
const int bluePin = 11;  // Pin untuk LED biru

// Mendefinisikan ukuran keypad
const byte ROWS = 4; // 4 baris
const byte COLS = 4; // 4 kolom

// Mendefinisikan pin untuk baris dan kolom
byte rowPins[ROWS] = {9, 8, 7, 6}; // Pin baris (R1, R2, R3, R4)
byte colPins[COLS] = {5, 4, 3, 2}; // Pin kolom (C1, C2, C3, C4)

// Mendefinisikan karakter pada keypad
char keys[ROWS][COLS] = {
  {'1', '2', '3', 'A'},
  {'4', '5', '6', 'B'},
  {'7', '8', '9', 'C'},
  {'*', '0', '#', 'D'}
};

// Membuat objek Keypad
Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

void setup() {
  Serial.begin(9600); // Memulai komunikasi Serial
  
  // Mengatur pin LED sebagai output
  pinMode(redPin, OUTPUT);
  pinMode(greenPin, OUTPUT);
  pinMode(bluePin, OUTPUT);
  
  // Mematikan semua LED pada awalnya
  digitalWrite(redPin, LOW);
  digitalWrite(greenPin, LOW);
  digitalWrite(bluePin, LOW);
}

void loop() {
  char key = keypad.getKey(); // Membaca input dari keypad

  if (key) { // Jika ada tombol yang ditekan
    Serial.print("Tombol yang ditekan: "); // Tampilkan tombol yang ditekan
    Serial.println(key); // Menampilkan karakter yang ditekan di Serial Monitor
    
    switch (key) {
      case '1': // Nyalakan LED merah
        digitalWrite(redPin, HIGH);
        Serial.println("LED Merah ON"); // Keterangan LED Merah nyala
        break;
      case '2': // Nyalakan LED hijau
        digitalWrite(greenPin, HIGH);
        Serial.println("LED Hijau ON"); // Keterangan LED Hijau nyala
        break;
      case '3': // Nyalakan LED biru
        digitalWrite(bluePin, HIGH);
        Serial.println("LED Biru ON"); // Keterangan LED Biru nyala
        break;
      case '4': // Matikan LED merah
        digitalWrite(redPin, LOW);
        Serial.println("LED Merah OFF"); // Keterangan LED Merah mati
        break;
      case '5': // Matikan LED hijau
        digitalWrite(greenPin, LOW);
        Serial.println("LED Hijau OFF"); // Keterangan LED Hijau mati
        break;
      case '6': // Matikan LED biru
        digitalWrite(bluePin, LOW);
        Serial.println("LED Biru OFF"); // Keterangan LED Biru mati
        break;
      default:
        // Tambahkan kasus lain jika diperlukan
        break;
    }
  }
}
